! =====================================================================
! R2-FILIAL  (Cisco 2911)
! LAN da filial, DHCP, OSPF e SSH
! Colar em: enable -> configure terminal
! =====================================================================
enable
configure terminal
hostname R2-FILIAL
no ip domain-lookup
enable secret Cisco123!
service password-encryption
banner motd # Acesso apenas a pessoal autorizado #

! ---- SSH ----
ip domain-name lab.local
username admin privilege 15 secret Cisco123!
crypto key generate rsa general-keys modulus 1024
ip ssh version 2
line vty 0 4
 transport input ssh
 login local
 exec-timeout 10 0
exit
line console 0
 logging synchronous
 exec-timeout 10 0
exit

! ---- LAN da filial (ligada a uma porta de acesso VLAN 30 do SW2) ----
interface GigabitEthernet0/0
 description LAN FILIAL (SW2-FILIAL)
 ip address 192.168.30.1 255.255.255.0
 no shutdown
exit

! ---- Ligação para a sede ----
interface GigabitEthernet0/1
 description Ligacao WAN para R1-SEDE
 ip address 10.0.0.2 255.255.255.252
 no shutdown
exit

! ---- DHCP ----
ip dhcp excluded-address 192.168.30.1 192.168.30.10

ip dhcp pool VLAN30-FILIAL
 network 192.168.30.0 255.255.255.0
 default-router 192.168.30.1
 dns-server 192.168.10.10
exit

! ---- OSPF ----
router ospf 1
 router-id 2.2.2.2
 network 192.168.30.0 0.0.0.255 area 0
 network 10.0.0.0 0.0.0.3 area 0
 passive-interface GigabitEthernet0/0
exit

end
write memory
